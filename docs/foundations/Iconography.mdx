import { Meta } from '@storybook/blocks';
import { useEffect, useState, useMemo } from 'react';

<Meta title="Foundations/Iconography" />

export const ICONS_REPO = 'UN-OCHA/humanitarian-icons-2026-BDU';
export const CDN_BASE = `https://cdn.jsdelivr.net/gh/${ICONS_REPO}@main`;
export const METADATA_URL = `${CDN_BASE}/metadata.json`;
export const SVG_URL = (key) => `${CDN_BASE}/svg/${key}.svg`;

export const IconGallery = () => {
  const [meta, setMeta] = useState(null);
  const [err, setErr] = useState(null);
  const [q, setQ] = useState('');
  const [family, setFamily] = useState('All');

  useEffect(() => {
    fetch(METADATA_URL)
      .then((r) => {
        if (!r.ok) throw new Error(`HTTP ${r.status}`);
        return r.json();
      })
      .then(setMeta)
      .catch((e) => setErr(String(e)));
  }, []);

  const filtered = useMemo(() => {
    if (!meta) return [];
    const entries = Object.entries(meta.icons);
    const qlc = q.trim().toLowerCase();
    return entries.filter(([key, v]) => {
      if (family !== 'All' && v.family !== family) return false;
      if (!qlc) return true;
      return (
        key.toLowerCase().includes(qlc) ||
        (v.name || '').toLowerCase().includes(qlc) ||
        (v.family || '').toLowerCase().includes(qlc)
      );
    });
  }, [meta, q, family]);

  if (err) {
    return (
      <div style={{ padding: 16, border: '1px solid #F9C0C5', background: '#FEF2F3', color: '#AB1D37', fontSize: 14 }}>
        Couldn't load icon metadata from jsDelivr ({err}). The live gallery requires an internet connection. Browse the icons directly at{' '}
        <a href={`https://github.com/${ICONS_REPO}/tree/main/svg`} style={{ color: '#0077B8' }}>
          github.com/{ICONS_REPO}
        </a>.
      </div>
    );
  }
  if (!meta) {
    return <div style={{ padding: 16, color: '#737373', fontSize: 14 }}>Loading {389}+ icons from jsDelivr…</div>;
  }

  const families = ['All', ...meta.families];

  return (
    <div>
      <div style={{
        display: 'flex',
        gap: 12,
        alignItems: 'center',
        flexWrap: 'wrap',
        marginBottom: 20,
        padding: '16px 0',
        borderBottom: '1px solid #E5E5E5',
      }}>
        <input
          type="search"
          placeholder="Search icons…"
          value={q}
          onChange={(e) => setQ(e.target.value)}
          style={{
            flex: '1 1 240px',
            padding: '8px 12px',
            fontFamily: 'Roboto, Arial, sans-serif',
            fontSize: 14,
            border: '1px solid #BFBFBF',
            borderRadius: 0,
            background: '#FFFFFF',
            color: '#161616',
          }}
        />
        <select
          value={family}
          onChange={(e) => setFamily(e.target.value)}
          style={{
            padding: '8px 12px',
            fontFamily: 'Roboto, Arial, sans-serif',
            fontSize: 14,
            border: '1px solid #BFBFBF',
            background: '#FFFFFF',
            color: '#161616',
            borderRadius: 0,
          }}
        >
          {families.map((f) => <option key={f} value={f}>{f}</option>)}
        </select>
        <span style={{ fontSize: 13, color: '#737373', fontFamily: 'Roboto Mono, monospace' }}>
          {filtered.length} / {Object.keys(meta.icons).length}
        </span>
      </div>

      <div style={{
        display: 'grid',
        gridTemplateColumns: 'repeat(auto-fill, minmax(120px, 1fr))',
        gap: 1,
        background: '#E5E5E5',
        border: '1px solid #E5E5E5',
      }}>
        {filtered.map(([key, v]) => (
          <div
            key={key}
            title={`${v.name} — ${v.family}`}
            style={{
              background: '#FFFFFF',
              padding: '16px 12px 10px',
              display: 'flex',
              flexDirection: 'column',
              alignItems: 'center',
              gap: 8,
              minHeight: 110,
            }}
          >
            <img
              src={SVG_URL(key)}
              alt={v.name}
              loading="lazy"
              style={{ width: 40, height: 40, objectFit: 'contain' }}
            />
            <div style={{
              fontSize: 11,
              lineHeight: 1.3,
              color: '#525252',
              fontFamily: 'Roboto, Arial, sans-serif',
              textAlign: 'center',
              wordBreak: 'break-word',
            }}>{v.name}</div>
          </div>
        ))}
      </div>

      {filtered.length === 0 && (
        <div style={{ padding: 32, textAlign: 'center', color: '#737373', fontSize: 14 }}>
          No icons match "{q}" in family "{family}".
        </div>
      )}
    </div>
  );
};

# Iconography

OCHA products draw on two icon libraries, each in a defined role. The decision is governed by **how the user encounters the icon — reading it, or acting on it.**

## Three roles

### 1. UI chrome → Font Awesome (free)

Anything that signals action, state, or affordance: buttons, navigation, toolbars, banners, filters, modals, status indicators, close / back / refresh, menus, search. Glanced at and clicked, not studied. Font Awesome is optically tuned for that role and includes a **Humanitarian category** that mirrors the OCHA Humanitarian set — OCHA co-built it.

### 2. Content & pictograms → OCHA Humanitarian Icons

Anything where the icon is what the user is reading: KPI cards, pictogram charts, chart category markers, illustrations in reports and sit-reps, infographic callouts, empty-state illustrations, legends and keys that explain symbology. **A reading icon can still be clickable** — a Health icon inside a KPI card that opens a detail view is still read first. It stays OCHA Humanitarian.

### 3. Maps → OCHA cartographic conventions

Start with OCHA's standard symbology. When an icon sits on top of a map:

- Bigger maps with room → **OCHA Humanitarian Icons**.
- Small maps (dashboard tiles, sit-rep insets, mini-maps) → **Font Awesome** — OCHA icons become dense at that scale.
- If Font Awesome lacks the humanitarian concept, fall back to OCHA Humanitarian at the smallest tolerable size, or use a generic pin + label.

### Bonus — partner and social logos → Font Awesome Brands

X, LinkedIn, YouTube, GitHub, Slack, partner-organisation marks. Font Awesome Brands covers these, free.

## Implementation

### OCHA Humanitarian Icons

Follow the README in [`UN-OCHA/humanitarian-icons-2026-BDU`](https://github.com/UN-OCHA/humanitarian-icons-2026-BDU). Quickest path is to load any icon as an SVG via the jsDelivr CDN and recolour it with `currentColor` or a CSS `mask`. The repo also ships an icon-font build for legacy and PowerPoint use, plus raw SVGs for Illustrator and other design tools.

### Font Awesome

Follow [fontawesome.com/docs](https://fontawesome.com/docs). Use the free CDN or self-host.

- Stay on the **Solid** weight, monochrome, recoloured to the OCHA palette.
- **Pin a specific Font Awesome version** in your stylesheets so visual drift doesn't sneak in across releases.
- For partner and social logos use the **Brands** weight.

## Accessibility

- Provide meaningful **`alt` text** on `<img>` or an `aria-label` on inline SVG. Decorative icons (e.g. a leading bullet) should use `alt=""` or `aria-hidden="true"`.
- **Never rely on icon alone** to convey meaning — pair it with text, a label, or a tooltip the first time a humanitarian concept appears in a product.
- Verify the icon's colour against its background meets **3:1 contrast** for graphic elements (WCAG 2.1 SC 1.4.11).
